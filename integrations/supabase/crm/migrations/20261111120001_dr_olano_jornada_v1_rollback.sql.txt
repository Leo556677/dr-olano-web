begin
  if exists (
    select 1 from public.citas a join public.citas b
      on a.recurso_id = b.recurso_id and a.id < b.id
     and tstzrange(a.inicio, a.fin, '[)') && tstzrange(b.inicio, b.fin, '[)')
    where a.estado in ('pendiente', 'confirmada') and b.estado in ('pendiente', 'confirmada')
  ) then
    raise notice 'Hay citas simultáneas en un mismo recurso: se conserva la restricción modificada. Cancela esas citas y vuelve a correr este paso.';
  else
    alter table public.citas drop constraint if exists citas_recurso_no_solapado;
    alter table public.citas add constraint citas_recurso_no_solapado
      exclude using gist (recurso_id with =, tstzrange(inicio, fin, '[)') with &&)
      where (recurso_id is not null and estado = any (array['pendiente', 'confirmada']));
  end if;
end
